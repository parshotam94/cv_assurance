# VisionTrust Assurance: Report & Finding JSON Schema Specification

This document defines the formal schema specification for compiled assurance reports, finding payloads, and cryptographic provenance records generated by VisionTrust Assurance.

---

## 1. Assurance Report JSON Schema (`AssuranceReportData`)

```json
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "AssuranceReportData",
  "type": "object",
  "required": [
    "report_id",
    "title",
    "timestamp",
    "version",
    "framework_version",
    "overall_risk_score",
    "overall_risk_level",
    "overall_disposition",
    "executive_summary",
    "threat_coverage",
    "findings"
  ],
  "properties": {
    "report_id": {
      "type": "string",
      "description": "Unique assurance dossier identifier (e.g. REP-8AF3...)"
    },
    "title": {
      "type": "string",
      "description": "Human-readable assessment title"
    },
    "timestamp": {
      "type": "string",
      "format": "date-time",
      "description": "UTC ISO-8601 compilation timestamp"
    },
    "version": {
      "type": "string",
      "description": "Report schema version (e.g. 1.0.0)"
    },
    "framework_version": {
      "type": "string",
      "description": "VisionTrust Assurance software release version"
    },
    "overall_risk_score": {
      "type": "number",
      "minimum": 0.0,
      "maximum": 100.0,
      "description": "Aggregated risk score across evaluated assets"
    },
    "overall_risk_level": {
      "type": "string",
      "enum": ["LOW", "MEDIUM", "HIGH", "CRITICAL"]
    },
    "overall_disposition": {
      "type": "string",
      "enum": ["ACCEPT", "REVIEW", "QUARANTINE"]
    },
    "executive_summary": {
      "type": "object",
      "properties": {
        "statement": { "type": "string" },
        "evaluated_datasets_count": { "type": "integer" },
        "evaluated_models_count": { "type": "integer" },
        "total_findings_count": { "type": "integer" },
        "critical_findings_count": { "type": "integer" },
        "high_findings_count": { "type": "integer" }
      },
      "required": ["statement", "total_findings_count"]
    },
    "threat_coverage": {
      "type": "array",
      "items": { "$ref": "#/$defs/CoverageItem" }
    },
    "findings": {
      "type": "array",
      "items": { "$ref": "#/$defs/FindingItem" }
    },
    "metadata": {
      "type": "object",
      "additionalProperties": true
    }
  },
  "$defs": {
    "CoverageItem": {
      "type": "object",
      "required": ["pillar_id", "threat_name", "status", "detection_method", "confidence"],
      "properties": {
        "pillar_id": { "type": "string" },
        "threat_name": { "type": "string" },
        "status": { "type": "string", "enum": ["COVERED", "NOT_TESTED", "DEGRADED"] },
        "detection_method": { "type": "string" },
        "confidence": { "type": "number", "minimum": 0.0, "maximum": 1.0 },
        "summary": { "type": "string" }
      }
    },
    "FindingItem": {
      "type": "object",
      "required": [
        "finding_id",
        "asset_type",
        "asset_id",
        "category",
        "severity",
        "confidence",
        "risk_score",
        "title",
        "description",
        "evidence",
        "detection_method",
        "recommendation"
      ],
      "properties": {
        "finding_id": { "type": "string" },
        "asset_type": { "type": "string", "enum": ["DATASET", "MODEL", "INFERENCE", "SYSTEM"] },
        "asset_id": { "type": "string" },
        "category": { "type": "string" },
        "severity": { "type": "string", "enum": ["LOW", "MEDIUM", "HIGH", "CRITICAL"] },
        "confidence": { "type": "number", "minimum": 0.0, "maximum": 1.0 },
        "risk_score": { "type": "number", "minimum": 0.0, "maximum": 100.0 },
        "title": { "type": "string" },
        "description": { "type": "string" },
        "evidence": { "type": "array", "items": { "type": "string" } },
        "affected_samples": { "type": "array", "items": { "type": "string" } },
        "detection_method": { "type": "string" },
        "limitations": { "type": "array", "items": { "type": "string" } },
        "recommendation": { "type": "string", "enum": ["ACCEPT", "REVIEW", "QUARANTINE"] }
      }
    }
  }
}
```

## 2. Cryptographic Provenance Record Schema (`ProvenanceRecord`)

```json
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "ProvenanceRecord",
  "type": "object",
  "required": [
    "record_id",
    "input_sha256",
    "model_sha256",
    "preprocessing_sha256",
    "config_sha256",
    "output_sha256",
    "timestamp",
    "nonce",
    "sequence",
    "record_hash",
    "signature",
    "public_key_hex"
  ],
  "properties": {
    "record_id": { "type": "string" },
    "input_sha256": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "model_sha256": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "preprocessing_sha256": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "config_sha256": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "output_sha256": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "timestamp": { "type": "string", "format": "date-time" },
    "nonce": { "type": "string" },
    "sequence": { "type": "integer", "minimum": 1 },
    "record_hash": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "signature": { "type": "string", "pattern": "^[a-f0-9]{128}$" },
    "public_key_hex": { "type": "string", "pattern": "^[a-f0-9]{64}$" },
    "raw_output": { "type": "object" }
  }
}
```
